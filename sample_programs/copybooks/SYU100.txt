      *--------------------------------------------------------------*          
      *    BUBBLE SORT: BINARY SORT ONLY !!!!                                   
      *--------------------------------------------------------------*          
      * 02/11/03 MCV NEW                                                MCV03022
      *--------------------------------------------------------------*          
                                                                                
      *    BUBBLE1 - INDEX                                                      
      *    BUBBLE2 - INDEX                                                      
      *    ENTRY   - ARRAY OCCURRANCE.                                          
      *    SWAP    - SWAP SAVE AREA.  LENGTH MUST BE >= SORT ELEMENT            
                                                                                
           EVALUATE TRUE                                                        
              WHEN LENGTH OF :ENTRY: (1) > LENGTH OF :SWAP:                     
                   SET WS-OLS-ERROR-ENTRY-LENGTH     TO  TRUE                   
              WHEN WS-OLS-KEY-LENGTH <= ZERO                                    
                   SET WS-OLS-ERROR-KEY-LENGTH       TO  TRUE                   
              WHEN WS-OLS-KEY-POSITION <= ZERO                                  
                   SET WS-OLS-ERROR-KEY-POSITION     TO  TRUE                   
              WHEN WS-OLS-SORT-BINARY-ASC                                       
              WHEN WS-OLS-SORT-BINARY-DES                                       
                   SET WS-OLS-RETURN-OK              TO  TRUE                   
              WHEN OTHER                                                        
                   SET WS-OLS-ERROR-SEQ-CODE-INVALID TO  TRUE                   
           END-EVALUATE                                                         
                                                                                
           IF WS-OLS-RETURN-OK                                                  
              PERFORM VARYING :BUBBLE1:                                         
                      FROM    +1                                                
                      BY      +1                                                
                      UNTIL   :BUBBLE1: >= WS-OLS-NUMBER-ENTRIES                
                 COMPUTE :BUBBLE2: = :BUBBLE1: + 1                              
                 PERFORM VARYING :BUBBLE2:                                      
                         FROM    :BUBBLE2:                                      
                         BY      +1                                             
                         UNTIL   :BUBBLE2: > WS-OLS-NUMBER-ENTRIES              
                    IF ( WS-OLS-SORT-BINARY-ASC       AND                       
                          :ENTRY: (:BUBBLE1:)                                   
                             (WS-OLS-KEY-POSITION : WS-OLS-KEY-LENGTH)          
                        > :ENTRY: (:BUBBLE2:)                                   
                             (WS-OLS-KEY-POSITION : WS-OLS-KEY-LENGTH))         
                      OR                                                        
                       ( WS-OLS-SORT-BINARY-DES       AND                       
                          :ENTRY: (:BUBBLE1:)                                   
                             (WS-OLS-KEY-POSITION : WS-OLS-KEY-LENGTH)          
                        < :ENTRY: (:BUBBLE2:)                                   
                             (WS-OLS-KEY-POSITION : WS-OLS-KEY-LENGTH))         
                                                                                
                       MOVE :ENTRY: (:BUBBLE1:)  TO  :SWAP:                     
                       MOVE :ENTRY: (:BUBBLE2:)  TO  :ENTRY: (:BUBBLE1:)        
                       MOVE :SWAP:               TO  :ENTRY: (:BUBBLE2:)        
                    END-IF                                                      
                END-PERFORM                                                     
              END-PERFORM.
