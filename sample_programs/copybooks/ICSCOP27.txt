       PROCESS-NET-PRICE-ORDR-TYPE.                                             
                                                                                
           IF ORHOF-NP-ORDER-IND > SPACE                                        
              EVALUATE TRUE                                                     
                  WHEN ORHOF-NP-ORDER-BACK-2-BACK                               
                  WHEN ORHOF-NP-ORDER-REPLNISHMNT                               
                  WHEN ORHOF-NP-TREAT-AS-STOCK                          TGF21281
                       MOVE ORHOF-NP-ORDER-IND                                  
                                       TO ICWCOP-NP-ORDER-IND                   
                  WHEN OTHER                                                    
                       MOVE ORSH-ORDER-TYPE                                     
                                       TO ICWCOP-NP-ORDER-IND                   
              END-EVALUATE                                                      
           ELSE                                                                 
              IF ORSH-ORDER-TYPE > SPACES                                       
                 MOVE ORSH-ORDER-TYPE  TO ICWCOP-NP-ORDER-IND                   
              ELSE                                                              
                 IF OET-ORDER-TYPE > SPACES                                     
                    MOVE OET-ORDER-TYPE                                         
                                       TO ICWCOP-NP-ORDER-IND                   
                 ELSE                                                           
                   MOVE SPACE          TO ICWCOP-NP-ORDER-IND                   
                 END-IF                                                         
              END-IF                                                    SYJ21218
           END-IF.                                                              
                                                                                
           IF ICWCOP-NP-DIRECT-ORDER  AND                               SYJ22280
              ORHOF-SAP-FG-ORD                                          SYJ22280
              MOVE ORHOF-SAP-FG-ORD-SW TO ICWCOP-NP-ORDER-IND           SYJ22280
           END-IF.                                                      SYJ22280
      *                                                                         
           IF ICWCOP-NP-DIRECT-ORDER                                    SYJ21218
           OR ORHOF-NP-ORDER-BACK-2-BACK                                SYJ21218
           OR ORHOF-NP-ORDER-REPLNISHMNT                                SYJ21218
           OR ORHOF-SAP-FG-ORD                                          SYJ22280
              CONTINUE                                                  SYJ21218
           ELSE                                                         SYJ21218
              MOVE 'S'                 TO ICWCOP-NP-ORDER-IND           SYJ21218
           END-IF.                                                      SYJ21218
                                                                        SYJ21218
       NET-PRICE-ORDR-TYP-EXIT.                                                 
           EXIT.
