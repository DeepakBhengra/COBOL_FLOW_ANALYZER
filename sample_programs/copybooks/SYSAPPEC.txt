       990-APP-ERR-MSG.                                                         
                                                                                
           MOVE '***APPLICATION ERROR MSG FOR PROGRAM '                         
                                       TO SYWAPPER-ERROR-MSG1                   
           MOVE ' ,TRAN:'              TO SYWAPPER-ERROR-MSG2                   
           MOVE LENGTH OF SYWAPPER-ERROR-MSG                                    
                                       TO SYWAPPER-MSG-LEN                      
      *                                                                         
           EXEC CICS WRITEQ TD                                                  
                     QUEUE  ('CSSL')                                            
                     FROM   (SYWAPPER-ERROR-MSG)                                
                     LENGTH (SYWAPPER-MSG-LEN)                                  
                     NOHANDLE                                                   
           END-EXEC.                                                            
      *                                                                         
           IF SYWAPPER-MESSAGE-1 > SPACES                                       
              MOVE LENGTH OF SYWAPPER-MESSAGE-1                                 
                                       TO SYWAPPER-MSG-LEN                      
              EXEC CICS WRITEQ TD                                               
                        QUEUE  ('CSSL')                                         
                        FROM   (SYWAPPER-MESSAGE-1)                             
                        LENGTH (SYWAPPER-MSG-LEN)                               
                        NOHANDLE                                                
              END-EXEC                                                          
           END-IF.                                                              
      *                                                                         
           IF SYWAPPER-MESSAGE-2 > SPACES                                       
              MOVE LENGTH OF SYWAPPER-MESSAGE-2                                 
                                       TO SYWAPPER-MSG-LEN                      
              EXEC CICS WRITEQ TD                                               
                        QUEUE  ('CSSL')                                         
                        FROM   (SYWAPPER-MESSAGE-2)                             
                        LENGTH (SYWAPPER-MSG-LEN)                               
                        NOHANDLE                                                
              END-EXEC                                                          
           END-IF.                                                              
      *                                                                         
           IF SYWAPPER-MESSAGE-3 > SPACES                                       
              MOVE LENGTH OF SYWAPPER-MESSAGE-3                                 
                                       TO SYWAPPER-MSG-LEN                      
              EXEC CICS WRITEQ TD                                               
                        QUEUE  ('CSSL')                                         
                        FROM   (SYWAPPER-MESSAGE-3)                             
                        LENGTH (SYWAPPER-MSG-LEN)                               
                        NOHANDLE                                                
              END-EXEC                                                          
           END-IF.

       990-APP-ERR-MSG-EXIT.                                                    
                       EXIT.
