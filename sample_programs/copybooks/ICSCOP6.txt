       A006-010-READ-ORTCNPT.                                                   
      *                                                                         
           EXEC SQL                                                             
                SELECT  GOP_PRG_TYP                                             
                  INTO :ORTCNPT-GOP-PRG-TYP                                     
                  FROM  OR_CONTRACT_PART                                        
                 WHERE CO_CD     = :ORTCNPT-CO-CD                               
                   AND BR_NBR    = :ORTCNPT-BR-NBR                              
                   AND CUST_NBR  = :ORTCNPT-CUST-NBR                            
                   AND CUST_SFX  = :ORTCNPT-CUST-SFX                            
                   AND GSA_NBR   = :ORTCNPT-GSA-NBR                             
                   AND PART_NBR  = :ORTCNPT-PART-NBR                            
           END-EXEC.                                                            
      *                                                                         
           MOVE SQLCODE TO SQLCODE-STATUS.                                      
      *                                                                         
           IF SQLCODE-NORESP                                                    
           OR SQLCODE-NOT-FOUND                                                 
              NEXT SENTENCE                                                     
           ELSE                                                                 
              MOVE 'A061'               TO WS-ABCODE                            
              MOVE 'A006-010-READ-ORT ' TO SYWDB2-EXTRA-MESSAGE-1               
              PERFORM 990-SQL-MESSAGE                                           
              PERFORM 999-ABEND.                                                
      *                                                                         
       A006-020-READ-ORTCNVD.                                                   
      *                                                                         
           EXEC SQL                                                             
                SELECT  GOP_PRG_TYP                                             
                  INTO :ORTCNVD-GOP-PRG-TYP                                     
                  FROM  OR_CONTRACT_VENDOR                                      
                 WHERE CO_CD     = :ORTCNVD-CO-CD                               
                   AND BR_NBR    = :ORTCNVD-BR-NBR                              
                   AND CUST_NBR  = :ORTCNVD-CUST-NBR                            
                   AND CUST_SFX  = :ORTCNVD-CUST-SFX                            
                   AND GSA_NBR   = :ORTCNVD-GSA-NBR                             
                   AND VND_NBR   = :ORTCNVD-VND-NBR                             
           END-EXEC.                                                            
      *                                                                         
           MOVE SQLCODE TO SQLCODE-STATUS.                                      
      *                                                                         
           IF SQLCODE-NORESP                                                    
           OR SQLCODE-NOT-FOUND                                                 
              NEXT SENTENCE                                                     
           ELSE                                                                 
              MOVE 'A062'               TO WS-ABCODE                            
              MOVE 'A006-020-READ-ORT ' TO SYWDB2-EXTRA-MESSAGE-1               
              PERFORM 990-SQL-MESSAGE                                           
              PERFORM 999-ABEND.
