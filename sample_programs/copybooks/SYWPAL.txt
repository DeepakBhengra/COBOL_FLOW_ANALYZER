      * CURSOR SYXPAL_WEIGHT_CSR.  THIS WILL GET ALL RECORDS IN                 
      * DATABASE FOR THIS ORDER.

           EXEC SQL                                                             
               DECLARE SYXPAL_WEIGHT_CSR  CURSOR FOR                            
                   SELECT PKG_SEQ_NBR,                                          
                          EST_LENGTH,                                           
                          EST_WIDTH,                                            
                          EST_HEIGHT,                                           
                          EST_VOL,                                              
                          DIM_UM,                                               
                          EST_WGT,                                              
                          WGT_UM,                                               
                          HEAVY_SKU_FLG,                                        
                          EST_OSZ_GIRTH_FLG,                            GXE18292
                          LAST_CHG_OPER_ID,                             GXE18292
                          LAST_CHG_PGM_ID,                              GXE18292
                          LST_CHG_TS,                                   GXE18292
                          CNTNR_TYP_CD,                                 GXE18292
                          HAZMAT_ID,                                    GXE18292
                          FRT_CLS                                       GXE18292
                   FROM SHIP_PAL_WEIGHT                                         
                   WHERE CO_CD        = :ORTSHPAL-CO-CD        AND              
                         BR_NBR       = :ORTSHPAL-BR-NBR       AND              
                         ORDR_NBR     = :ORTSHPAL-ORDR-NBR     AND              
                         DIST_NBR     = :ORTSHPAL-DIST-NBR     AND              
                         SHIP_NBR     = :ORTSHPAL-SHIP-NBR     AND              
                         ORDR_ENTY_DT = :ORTSHPAL-ORDR-ENTY-DT                  
                   ORDER BY PKG_SEQ_NBR                                         
                   FOR FETCH ONLY                                               
                   OPTIMIZE FOR 1 ROWS                                          
           END-EXEC.
