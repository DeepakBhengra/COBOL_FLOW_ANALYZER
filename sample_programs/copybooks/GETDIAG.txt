           EXEC SQL GET DIAGNOSTICS                                             
              :DIAG-ROW-COUNT     = ROW_COUNT ,                                 
              :DIAG-NUM-CONDNS    = NUMBER                                      
           END-EXEC                                                             
                                                                                
           PERFORM VARYING DIAG-X                                               
                FROM   +1                                                       
                BY     +1                                                       
                UNTIL  DIAG-X > DIAG-NUM-CONDNS                                 
                                                                                
              EXEC SQL GET DIAGNOSTICS CONDITION :DIAG-X                        
                 :DIAG-RET-SQLCODE     = DB2_RETURNED_SQLCODE ,                 
                 :DIAG-RET-SQLSTATE    = RETURNED_SQLSTATE    ,                 
                 :DIAG-ROW-NUM         = DB2_ROW_NUMBER       ,                 
                 :SYWDB2-ERROR-MESSAGE = MESSAGE_TEXT                           
              END-EXEC                                                          
                                                                                
              PERFORM DIAG-DISPLAY                                              
                                                                                
           END-PERFORM.
