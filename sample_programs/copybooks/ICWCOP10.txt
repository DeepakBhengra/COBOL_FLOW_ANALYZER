      * CURSOR FOR ORDER_SKU_ACOP_TOTALS ORDER CHECK                            
           EXEC SQL                                                             
                DECLARE ORDR_SKU_ACOP_TL_CURSOR CURSOR FOR                      
                SELECT  IM_PART_NBR                                             
                       ,ACOP_TRK_NBR                                            
                       ,TOT_ORDR_QTY                                            
                       ,TOT_BORDR_QTY                                           
                       ,TOT_SHIP_QTY                                            
                       ,ACOP_MIN_QTY                                            
                       ,ACOP_TIER_TYPE_CD                                       
                       ,VEND_NBR                                                
                  FROM  ORDER_SKU_ACOP_TOTALS                                   
                 WHERE CO_CD        = :ORTLPAS-CO-CD                            
                   AND ORDR_BR_NBR  = :ORTLPAS-ORDR-BR-NBR                      
                   AND ORDR_NBR     = :ORTLPAS-ORDR-NBR                         
                   AND ORDR_DT      = :ORTLPAS-ORDR-DT                          
                   AND CUST_BR_NBR  = :ORTLPAS-CUST-BR-NBR                      
                   AND CUST_NBR     = :ORTLPAS-CUST-NBR                         
                 ORDER BY ACOP_TRK_NBR                                          
           END-EXEC.                                                            
                                                                                
      * CURSOR FOR ORDER_SKU_ACOP_TOTALS PART DELETE                            
           EXEC SQL                                                             
                DECLARE ORDR_SKU_ACOP_DELT_CURSOR CURSOR FOR                    
                SELECT                                                          
                   ACOP_TRK_NBR                                                 
                  ,TOT_ORDR_QTY                                                 
                  ,TOT_BORDR_QTY                                                
                  ,TOT_SHIP_QTY                                                 
                  ,ACOP_MIN_QTY                                                 
                  ,ACOP_TIER_TYPE_CD                                            
                  ,INVC_DT                                                      
                FROM                                                            
                   ORDER_SKU_ACOP_TOTALS                                        
                WHERE                                                           
                   CO_CD             =  :ORTLPAS-CO-CD        AND               
                   ORDR_BR_NBR       =  :ORTLPAS-ORDR-BR-NBR  AND               
                   ORDR_NBR          =  :ORTLPAS-ORDR-NBR     AND               
                   ORDR_DT           =  :ORTLPAS-ORDR-DT      AND               
                   CUST_BR_NBR       =  :ORTLPAS-CUST-BR-NBR  AND               
                   CUST_NBR          =  :ORTLPAS-CUST-NBR                       
                FOR UPDATE OF                                                   
                   CO_CD                                                        
                  ,ORDR_BR_NBR                                                  
                  ,ORDR_NBR                                                     
                  ,ORDR_DT                                                      
                  ,IM_PART_NBR                                                  
                  ,VEND_NBR                                                     
                  ,ACOP_TRK_NBR                                                 
                  ,TOT_ORDR_QTY                                                 
                  ,TOT_BORDR_QTY                                                
                  ,TOT_SHIP_QTY                                                 
                  ,ACOP_MIN_QTY                                                 
                  ,ACOP_TIER_TYPE_CD                                            
           END-EXEC.
