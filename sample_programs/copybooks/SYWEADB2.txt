      *-----------------------------------------------------------*             
      * CURSOR DECLARATION FOR EA_BRANCH_SEQUENCING TABLE                       
      * TO FETCH NEAREST BRANCH BASED ON SHIP TO ZIP CODE FOR EA SKU            
      *-----------------------------------------------------------*             
           EXEC SQL                                                             
                DECLARE SW_BR_CURSOR CURSOR FOR                                 
                 SELECT VIRT_BR_NBR                                             
                  FROM EA_BRANCH_SEQUENCING                                     
                  WHERE CO_CD       = :ORTEABSQ-CO-CD                           
                    AND VEND_NBR    = :ORTEABSQ-VEND-NBR                        
                    AND FROM_ZIP   <= :ORTEABSQ-FROM-ZIP                        
                    AND TO_ZIP     >= :ORTEABSQ-TO-ZIP                          
                  ORDER BY VIRT_BR_PRTY                                         
                FOR FETCH ONLY                                                  
           END-EXEC.
