      *--------------------------------------------------------------*          
      * GET UNIQUE ORSHED + ORSHEDOF PATH CALL BY PRIMARY INDEX                 
      *--------------------------------------------------------------*          
      *    03/25/05  MCV NEW                                            MCV05087
                                                                                
           CALL 'CBLTDLI'           USING  GU                                   
                                           ORPCB                                
                                           ORH-ORSHED                           
                                           WS-ORSHED-SSA                        
                                           WS-ORSHEDOF-SSA.                     
                                                                                
           IF ORPCB-NORESP                                                      
              MOVE 'Y'                 TO  WS-ORSHEDOF-IO-SW                    
                                           WS-ORSHED-IO-SW                      
           ELSE                                                                 
               IF ORPCB-NOTFND                                                  
                   MOVE 'N'            TO  WS-ORSHEDOF-IO-SW                    
                   IF ORPCB-SEG-NAME = 'ORSHED  '                               
                       MOVE 'Y'        TO  WS-ORSHED-IO-SW                      
                   ELSE                                                         
                       MOVE 'N'        TO  WS-ORSHED-IO-SW                      
               ELSE                                                             
                   MOVE 'O701'         TO  WS-ABCODE                            
                   MOVE ORPCB-STS      TO  WS-STATUS                            
                   MOVE 'DL/1'         TO  WS-ERROR-CODE                        
                   MOVE SPACES         TO  WS-ERROR-OPTIONAL-DATA               
                   MOVE ORPCB          TO  WS-ERROR-DL1-PCB                     
                   MOVE WS-ABCODE      TO  WS-ERROR-DL1-ABCODE                  
                   PERFORM 999-ABEND.
