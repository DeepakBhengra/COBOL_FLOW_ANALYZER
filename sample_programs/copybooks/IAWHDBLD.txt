      *--- CURSOR DECLARATION TO HARD BUNDLE COMPONENT DB2 TABLE                
           EXEC SQL                                                             
                DECLARE HARD_BUNDLE_CSR CURSOR FOR                              
                SELECT  CMPNT_PART_NBR                                          
                       ,DESC_TXT                                                
                       ,UNIT_COST_AMT                                           
                       ,UNIT_PRC_AMT                                            
                       ,CMPNT_QTY                                               
                       ,HSN_SAC_CD                                              
                       ,PRCPL_CMPNT_IND                                         
                       ,LAST_CHG_TS                                             
                  FROM  HARD_BUNDLE_COMPONENT                                   
                 WHERE  CO_CD           = :INTHRDC-CO-CD                        
                   AND  PART_NBR        = :INTHRDC-PART-NBR                     
                   AND  CMPNT_PART_NBR  > :INTHRDC-CMPNT-PART-NBR               
              ORDER BY  PRCPL_CMPNT_IND DESC                                    
                       ,UNIT_PRC_AMT DESC                                       
           END-EXEC.
