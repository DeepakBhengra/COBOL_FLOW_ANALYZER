       999DS-CREATE-LINE-DESC.                                                  
      *                                                                         
           INITIALIZE WS-DS-DESC-VARIABLES.                                     
           INITIALIZE WS-DS-JUST-VARIABLES.                                     
      *                                                                         
      *--- LEFT JUSTIFY INPUT DESCRIPTION                                       
           MOVE WS-DS-IN-DESC-VAR         TO WS-DS-TAB-JUST-INDATA.             
           PERFORM 999DS-010-LEFT-JUSTIFY-INPUT.                                
           MOVE WS-DS-TAB-JUST-OUTDATA    TO WS-DS-DESC-VAR.                    
           MOVE WS-DS-IN-DESC-LIT         TO WS-DS-DESC-LIT.                    
           MOVE WS-DS-DESC                TO WS-DS-ORDDESC-LINES.               
      *                                                                         
           IF  WS-DS-IN-DESC-NBR1 > ZERO                                        
           AND WS-DS-DESC-VAR1    > SPACES                                      
           AND WS-ORSLNE-NORESP                                         TGF19067
               MOVE WS-DS-IN-DESC-NBR1    TO ORD-DESC-NBR                       
               MOVE WS-DS-ORDDESC-LINE1   TO ORD-DESCRIPTION-ORD                
               PERFORM 1344-ISRT-ORSDES                                         
           END-IF.                                                              
      *                                                                         
           IF  WS-DS-IN-DESC-NBR2 > ZERO                                        
           AND WS-DS-ORDDESC-LINE2 > SPACES                                     
           AND WS-ORSLNE-NORESP                                         TGF19067
               MOVE WS-DS-IN-DESC-NBR2    TO ORD-DESC-NBR                       
               MOVE WS-DS-ORDDESC-LINE2   TO ORD-DESCRIPTION-ORD                
               PERFORM 1344-ISRT-ORSDES                                         
           END-IF.                                                              
      *                                                                         
           IF  WS-DS-IN-DESC-NBR3 > ZERO                                        
           AND WS-DS-ORDDESC-LINE3 > SPACES                                     
           AND WS-ORSLNE-NORESP                                         TGF19067
               MOVE WS-DS-IN-DESC-NBR3    TO ORD-DESC-NBR                       
               MOVE WS-DS-ORDDESC-LINE3   TO ORD-DESCRIPTION-ORD                
               PERFORM 1344-ISRT-ORSDES                                         
           END-IF.                                                              
      *                                                                         
       999DS-010-LEFT-JUSTIFY-INPUT.                                            
      *                                                                         
           PERFORM                                                              
             VARYING WS-DS-SUBIN FROM 1 BY 1                                    
               UNTIL WS-DS-SUBIN > WS-DS-SUB-LIMIT                              
                  IF WS-DS-TAB-JUST-IN(WS-DS-SUBIN) > SPACES                    
                  OR WS-DS-MOVE-CHAR-OK                                         
                     MOVE 'Y' TO WS-DS-MOVE-CHAR-IND                            
                     ADD   1  TO WS-DS-SUBOUT                                   
                     MOVE WS-DS-TAB-JUST-IN (WS-DS-SUBIN)                       
                       TO WS-DS-TAB-JUST-OUT(WS-DS-SUBOUT)                      
                  END-IF                                                        
           END-PERFORM.
