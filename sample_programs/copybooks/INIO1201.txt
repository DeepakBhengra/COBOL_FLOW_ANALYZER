      *                                                                         
      *    INSHED + INSHEDOF PATH CALL VIA PART INDEX                           
      *    COMMON IOAREA USED IN THIS ROUTINE IS CPYBOOK INWPATH                
      *    CREATED: 01/06/04 MISSYS                                             
      *                                                                         
           CALL 'CBLTDLI' USING        GU                                       
                                       INPCB-PART                               
                                       COMMON-INDMST-IOAREA                     
                                       WS-INAL2-SSA                             
                                       WS-INSHEDOF-SSA                          
                                                                                
           IF INPCB-PART-NORESP                                                 
               MOVE 'Y'                TO  WS-INSHED-IO-SW                      
                                           WS-INSHEDOF-IO-SW                    
           ELSE                                                                 
               IF INPCB-PART-NOTFND                                             
                   MOVE 'N'            TO  WS-INSHEDOF-IO-SW                    
                   IF INPCB-PART-SEG-NAME = 'INSHED'                            
                       MOVE 'Y'        TO  WS-INSHED-IO-SW                      
                   ELSE                                                         
                       MOVE 'N'        TO  WS-INSHED-IO-SW                      
               ELSE                                                             
                   MOVE 'I201'         TO  WS-ABCODE                            
                   MOVE INPCB-PART-STS TO  WS-STATUS                            
                   MOVE 'DL/1'         TO  WS-ERROR-CODE                        
                   MOVE SPACES         TO  WS-ERROR-OPTIONAL-DATA               
                   MOVE INPCB-PART     TO  WS-ERROR-DL1-PCB                     
                   MOVE WS-ABCODE      TO  WS-ERROR-DL1-ABCODE                  
                   PERFORM 999-ABEND.
