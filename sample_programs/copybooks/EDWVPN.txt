      * CURSOR FOR FETCHING IM PART NBR FROM PL_PARTS_LOCATOR                   
           EXEC SQL                                                             
                DECLARE PARTS_CSR CURSOR FOR                                    
                SELECT  PART_NBR                                                
                  FROM  PL_PARTS_LOCATOR                                        
                 WHERE CO_CD                  = :EDWVPN-IP-CO-CD        KZC18097
                   AND SCR_EXT_VEND_PART_NBR  = :EDWVPN-IP-SCR-EXT-VPN  KZC18097
           END-EXEC.                                                            
                                                                                
      * CURSOR FOR FETCHING EXTENDED VPN FROM PL_PARTS_LOCATOR                  
           EXEC SQL                                                             
                DECLARE EXVPN_CSR CURSOR FOR                                    
                SELECT  EXT_VEND_PART_NBR                                       
                  FROM  PL_PARTS_LOCATOR                                        
                 WHERE CO_CD    = :EDWVPN-IP-CO-CD                              
                   AND PART_NBR = :EDWVPN-IP-IM-PART-NBR                        
           END-EXEC.
