                       OR (WS-PCB-FOUND-SW  = 'Y')                      JWO01182

               IF WS-PCB-TYPE = TB-PSB-TY-TYPE ( WS-PCB-SUB )           JWO01182
                   MOVE TB-PSB-TY-ROW ( WS-PCB-SUB ) TO WS-PCB-ROW      JWO01182
                   MOVE 'Y' TO WS-PCB-FOUND-SW                          JWO01182
               END-IF                                                   JWO01182

           END-PERFORM.                                                 JWO01182
      *                                                                 JWO01182
