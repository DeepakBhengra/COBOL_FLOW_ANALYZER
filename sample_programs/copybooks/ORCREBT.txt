           MOVE SPACES      TO WS-SKU-HAS-REBATE-SW                             
                               WS-INM-COST-CODE.                                
           MOVE +0          TO WS-INM-COST-REBATE-AMT                           
                               WS-INM-COST-REB-CUST-AMT                 KLM97245
                               WS-INM-COST-REB-LOCAL-AMT                KLM97245

      *--- SET WS-INM-QTY-INQUIRY TO 1                                          
           IF WS-INM-QTY-ORDERED = +0                                           
              MOVE +1       TO WS-INM-QTY-INQUIRE                               
           ELSE                                                                 
              MOVE +0       TO WS-INM-QTY-INQUIRE.                              
      *                                                                         
           IF NOT COST-PROMO-READ                                               
              MOVE 'Y'      TO WS-COST-PROMO-READ-SW                            
              IF  WS-INSTANT-REBATE-ALLOWED

       6100-CHECK-REBATE-RESTRICTION.                                           
      *                                                                         
      *--- CHECK STATUS                                                         
           IF INM-DELETE-STATUS                                                 
              GO TO 6100-EXIT.                                                  
      *                                                                         
      *--- CHECK EFFECTIVITY                                                    
      *Y2K********************************************************Y2K*  RXDY8308
      *      CA-FIX/2000 INSERTED THE FOLLOWING STATEMENT(S)         *  RXDY8308
      *Y2K********************************************************Y2K*  RXDY8308
           CALL 'CAWINUP'                                               RXDY8308
               USING INM-COST-EFF-DATE                                  RXDY8308
                     SYRPWSA-DEFAULT-WINDOW-SPEC                        RXDY8308
                     SYRPWSA-DECIMAL5-TYPE                              RXDY8308
                     SYRPWSA-INM-COST-EFF-DATE-Y2K                      RXDY8308
                     SYRPWSA-THIS-YEAR-Y2K                              RXDY8308
           CALL 'CAWINUP'                                               RXDY8308
               USING INM-COST-END-DATE                                  RXDY8308
                     SYRPWSA-DEFAULT-WINDOW-SPEC                        RXDY8308
                     SYRPWSA-DECIMAL5-TYPE                              RXDY8308
                     SYRPWSA-INM-COST-END-DATE-Y2K                      RXDY8308
                     SYRPWSA-THIS-YEAR-Y2K                              RXDY8308
      *Y2K********************************************************Y2K*  RXDY8308
      *      CA-FIX/2000 REPLACED THE FOLLOWING STATEMENT(S)         *  RXDY8308
      *Y2K********************************************************Y2K*  RXDY8308
      *    IF  INM-COST-EFF-DATE <= EIBDATE                             RXDY8308
           IF  SYRPWSA-INM-COST-EFF-DATE-Y2K <= EIBDATE                 RXDY8308
      *Y2K********************************************************Y2K*  RXDY8308
      *      CA-FIX/2000 REPLACED THE FOLLOWING STATEMENT(S)         *  RXDY8308
      *Y2K********************************************************Y2K*  RXDY8308
      *    AND INM-COST-END-DATE >= EIBDATE                             RXDY8308
           AND SYRPWSA-INM-COST-END-DATE-Y2K >= EIBDATE                 RXDY8308
               NEXT SENTENCE                                                    
           ELSE                                                                 
              GO TO 6100-EXIT.

      *--- ALLOW REBATE ONLY WHEN IT IS A SYSTEM COMPONENT                      
           IF INM-COST-SYS-COMPONENT-REBATE                                     
              IF NOT WS-INM-SYSTEM-REBATE-PROGRAM                               
                  GO TO 6100-EXIT.                                              
      *                                                                         
      *--- NO REBATES WILL BE APPLIED WHEN QTY LIMIT EXIST AND IR CUST          
      *--- SWITCH IS "L"                                                        
           IF WS-INSTANT-REBATE-LIMITED                                         
              IF INM-COST-MAX-QTY          > +0                                 
              OR INM-COST-MAQ              > +0                                 
              OR INM-COST-MIN-REQUIRED-QTY > +0                                 
                 GO TO 6100-EXIT.                                               
      *                                                                         
      *--- MAXIMUM QTY IS REACHED, REBATE WILL NOT BE APPLIED                   
           MOVE 'N'               TO  WS-INM-MAX-IND.                           
           IF  INM-COST-MAX-QTY > +0                                            
               MOVE 'IM-'  TO WS-INM-MAX-LIT                                    
               COMPUTE WS-INM-MAX   = INM-COST-MAX-QTY                          
                                    - INM-COST-MAX-QTY-TODATE                   
               IF INM-COST-MAX-QTY >= (INM-COST-MAX-QTY-TODATE +                
                                       WS-INM-QTY-INQUIRE      +                
                                       WS-INM-QTY-ORDERED)                      
                  MOVE 'Y'        TO WS-INM-MAX-IND                             
               ELSE                                                             
                  MOVE 'O'        TO WS-INM-MAX-IND                             
                  MOVE 'IM---'    TO WS-INM-WARNING-LIT                         
                  MOVE WS-INM-MAX TO WS-INM-WARNING-QTY                         
                  GO TO 6100-EXIT.                                              
      *                                                                         
      *--- MAQ IS REACHED, REBATE WILL NOT BE APPLIED                           
           MOVE 'N'               TO WS-INM-MAQ-IND.                            
           IF  INM-COST-MAQ > +0                                                
               PERFORM 6120-READ-CSFREC-ICP                                     
               MOVE 'CUST-'       TO WS-INM-MAQ-LIT                             
               COMPUTE WS-INM-MAQ =  INM-COST-MAQ                               
                                  -  MAQ-ICP-QTY-ORDERED                        
               IF INM-COST-MAQ   >= (MAQ-ICP-QTY-ORDERED +                      
                                     WS-INM-QTY-INQUIRE  +                      
                                     WS-INM-QTY-ORDERED)                        
                  MOVE 'Y'        TO WS-INM-MAQ-IND                             
               ELSE                                                             
                  MOVE 'O'        TO WS-INM-MAQ-IND                             
                  MOVE 'CUST-'    TO WS-INM-WARNING-LIT                         
                  MOVE WS-INM-MAQ TO WS-INM-WARNING-QTY                         
                  GO TO 6100-EXIT.                                              
      *                                                                         
      *--- MIN QTY IS NOT REACHED, REBATE WILL NOT BE APPLIED                   
           MOVE 'N'               TO WS-INM-MIN-IND.                            
           IF  INM-COST-MIN-REQUIRED-QTY > +0                                   
               MOVE 'MIN-'        TO WS-INM-MIN-LIT                             
               MOVE INM-COST-MIN-REQUIRED-QTY TO WS-INM-MIN                     
               IF WS-INM-QTY-INQUIRE > +0                                       
                  MOVE 'Y'        TO WS-INM-MIN-IND                             
               ELSE                                                             
               IF INM-COST-MIN-REQUIRED-QTY <= WS-INM-QTY-ORDERED               
                                             + WS-INM-QTY-ORDERED-ORIG          
                  MOVE 'Y'        TO WS-INM-MIN-IND                             
               ELSE                                                             
                  MOVE 'O'        TO WS-INM-MIN-IND                             
                  MOVE 'MIN--'    TO WS-INM-WARNING-LIT                         
                  MOVE WS-INM-MIN TO WS-INM-WARNING-QTY                         
                  GO TO 6100-EXIT.
