           CALL 'CBLTDLI' USING        GHNP                                     
                                       ORPCB2                                   
                                       ORSHOF-ORSSHHOF                          
                                       WS-ORSHED2-SSA                           
                                       WS-ORSSHH2-SSA                           
                                       WS-ORSSHHOF2-SSA.                        
           IF ORPCB2-NORESP                                                     
               MOVE 'Y'                TO  WS-ORSSHHOF-IO-SW                    
           ELSE                                                                 
               IF ORPCB2-NOTFND                                                 
                   MOVE 'N'            TO  WS-ORSSHHOF-IO-SW                    
               ELSE                                                             
                   IF ORPCB2-ENDFLE                                             
                       MOVE 'E'        TO  WS-ORSSHHOF-IO-SW                    
                   ELSE                                                         
                      MOVE '2277'     TO  WS-ABCODE                             
                       MOVE ORPCB2-STS TO  WS-STATUS                            
                       MOVE 'DL/1'     TO  WS-ERROR-CODE                        
                       MOVE SPACES     TO  WS-ERROR-OPTIONAL-DATA               
                       MOVE ORPCB2     TO  WS-ERROR-DL1-PCB                     
                       MOVE WS-ABCODE  TO  WS-ERROR-DL1-ABCODE                  
                       PERFORM 999-ABEND.
