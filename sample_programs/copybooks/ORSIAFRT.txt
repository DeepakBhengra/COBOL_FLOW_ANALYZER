      *   THIS COPYBOOK USED TO CALCULATE THE INDIA FREIGHT FOR THE    *        
      *   SALES ORDER AND INVOKED IN ORP560MD AND RWP566IM MODULES.    *        
      *                                                                         
      *   TO CALCULATE THE FREIGT SYSTEM CHECKS FOLLOWING PARAMETERS            
      *   1)CARRIER CODE                                                        
      *   2)PINCODE IN SHIP TO ADDRESS                                          
      *   3)BILLING BRANCH FOR REVENUE TABLE                                    
      *   4)SHIP BRANCH FOR COST TABLE                                          
      *   **************************************************************        
      *   1) BASED ON CUSTOMER SHIP TO PINCODE SYSTEM WILL CHECK                
      *      CORRESPOND ZONE USING CARRIER ZONES TABLE                          
      *   2) WITH THE ZONE, SYSTEM WILL CHECK FOR FREIGH COST FOR THE           
      *      CARRIER CODE AND REVENUE COST FOR THE CUSTOMER                     
      *   3) THERE ARE 9 FREIGHT RULES OUT OF WHICH ONLY 4 ARE ACTIVE           
      *      RULE 0 - FREIGHT COST WILL BE ZERO                                 
      *      RULE 1 - SINGLE RATE COSTING BASED ON WEIGHT/VALUE                 
      *      RULE 2 - AGGREGATE COSTING BASED ON ON WEIGHT/VALUE                
      *      RULE 3 - SINGLE WITH FLAT CHARGE                                   
      *                                                                         
      *      IN EXAMPLE HERE THE FREIGHT TILL FIRST 10 KG WILL BE RS.250        
      *      NEXT PER KG IT WILL BE 3 PER KG                                    
      *                                                                         
      *      IF THE WEIGHT IS 14KG FREIGHT WILL BE AS BELOW                     
      *      10 KG - 250                                                        
      *      FOR NEXT 4 KG IT WILL BE                                           
      *      14-10 = 4KG * RS 3 = 12                                            
      *      SO THE TOTAL FREIGHT FOR ABOVE WILL BE RS.262/-                    
      *                                                                         
      *      SIMILAR CALCULATION WILL APPLIES BASED ON STOCK SALES PRICE        
      **==============================================================**

                      ORL-FRT-CHARGE-AT-LINE  * ORGL-O-CESS2-RATE               
              COMPUTE WS-CESS2-AT-FRT-AMT-TOT       =                           
                      WS-CESS2-AT-FRT-AMT-TOT + ORGL-O-CESS2-AT-FRT-AMT         
           END-IF.                                                              
                                                                                
           IF ORGL-O-CESS3-YES                                                  
              IF ORL-SC-GST-TAX                                                 
                 COMPUTE ORGL-O-CESS3-AT-FRT-AMT    ROUNDED =                   
                        ((ORL-FRT-CHARGE-AT-LINE +                              
                          ORL-C-GST-AT-FRT       + ORL-S-GST-AT-FRT)            
                                                 * ORGL-O-CESS3-RATE)           
              ELSE                                                              
                 IF ORL-I-GST-TAX                                               
                    COMPUTE ORGL-O-CESS3-AT-FRT-AMT ROUNDED =                   
                        ((ORL-FRT-CHARGE-AT-LINE + ORL-I-GST-AT-FRT)            
                                                 * ORGL-O-CESS3-RATE)           
                 END-IF                                                         
              END-IF                                                            
              COMPUTE WS-CESS3-AT-FRT-AMT-TOT       =                           
                      WS-CESS3-AT-FRT-AMT-TOT + ORGL-O-CESS3-AT-FRT-AMT         
           END-IF.                                                              
                                                                                
           IF ORGL-O-CESS4-YES                                                  
              IF ORL-SC-GST-TAX                                                 
                 COMPUTE ORGL-O-CESS4-AT-FRT-AMT ROUNDED =                      
                         (ORL-C-GST-AT-FRT   + ORL-S-GST-AT-FRT)                
                                             * ORGL-O-CESS4-RATE                
              ELSE                                                              
                 IF ORL-I-GST-TAX                                               
                    COMPUTE ORGL-O-CESS4-AT-FRT-AMT ROUNDED =                   
                         (ORL-I-GST-AT-FRT   * ORGL-O-CESS4-RATE)               
                 END-IF                                                         
              END-IF                                                            
              COMPUTE WS-CESS4-AT-FRT-AMT-TOT       =                           
                     WS-CESS4-AT-FRT-AMT-TOT + ORGL-O-CESS4-AT-FRT-AMT          
           END-IF.
