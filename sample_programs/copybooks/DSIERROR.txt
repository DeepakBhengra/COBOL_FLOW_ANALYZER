      *---------------------------------------------------------------          
      *  DSI ERROR LIST                                                         
      *  TRANSLATE ERROR MESSAGES RETURNED FROM DSI LINK                        
      *---------------------------------------------------------------          
      * REVISIONS:                                                              
      * 12-20-99 TMH ADD CARRIER COD ERROR MESSAGE                      TMH99354
      * 08-13-96 TFM ADD CARRIER CONFLICT ERROR MESSAGE                 TFM96226
      * 07-27-96 TFM ADD ERROR MSG 14 FOR NON-OVERSIZED PRODUCTS        TFM96226
      * 07-11-95 GEH ADD ERROR MESSAGES 13, 25 & 30                     GEH95192
      * 07-05-95 MPB ADD ERR MSG 15 FOR OVERSIZED PRODUCT ORDERS        MPB95186
      * 07-03-95 GEH CHG MSG FOR ERROR CODE 113                         GEH95184
      *---------------------------------------------------------------          
           EVALUATE TSBR-RATED                                                  
             WHEN 0                                                             
                CONTINUE                                                        
             WHEN 11                                                            
                MOVE 'NOT A BASERATE CARRIER'   TO WS-DSI-ERRMSG                
             WHEN 13                                                            
                MOVE 'OT - CALL TRAFFIC DEPT'                           GEH95192
                                                TO WS-DSI-ERRMSG        GEH95192
             WHEN 14                                                    TFM96226
                MOVE 'PLEASE USE SML OR AIR CARR    '                   TFM96226
                                                TO WS-DSI-ERRMSG        TFM96226
             WHEN 15                                                    MPB95186
                MOVE 'OVERSIZED-USE LTL OR AIRH CARR'                   MPB95186
                                                TO WS-DSI-ERRMSG        MPB95186
             WHEN 16                                                    TMH99354
                MOVE 'SHIP COD ERROR      '                             TMH99354
                                                TO WS-DSI-ERRMSG        TMH99354
             WHEN 25                                                    GEH95192
                MOVE 'INVALID CARRIER FOR DESTINATION'                  GEH95192
                                                TO WS-DSI-ERRMSG        GEH95192
             WHEN 30                                                    GEH95192
                MOVE 'NO GROUND CARRIER AVAILABLE'                      GEH95192
                                                TO WS-DSI-ERRMSG        GEH95192
             WHEN 109                                                           
                MOVE 'INVALID NBR PRODUCTS'     TO WS-DSI-ERRMSG                
             WHEN 113                                                           
                MOVE 'NO PRODUCTS ASSIGNED TO DISTRIBUTION'             GEH95184
                                                TO WS-DSI-ERRMSG        GEH95184
             WHEN 115                                                           
                MOVE 'INVALID ORIGIN'           TO WS-DSI-ERRMSG                
             WHEN 116                                                           
                MOVE 'INVALID ZIP DESTINATION'  TO WS-DSI-ERRMSG                
             WHEN 321                                                   TFM96226
                MOVE 'ADDR/CARR CONFLICT-CHG CARRIER FIRST'             TFM96226
                                                TO WS-DSI-ERRMSG        TFM96226
             WHEN OTHER                                                         
                MOVE 'INVALID RATING: '         TO WS-DSI-ERRMSG                
                MOVE TSBR-RATED                 TO WS-DSI-ERRNUM                
                MOVE WS-DSI-ERRNUM              TO WS-DSI-ERRMSG(17:3)          
           END-EVALUATE.
