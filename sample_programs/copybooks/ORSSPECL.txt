      *    **********************************************************           
      *    *PURPOSE: ROUTINE TO TRANSLATE SPECIAL CHARACTERS       **           
      *    *         FOR PASS-THROUGH DATA ARCHIVE                 **           
      *    *                                                       **           
      *    *REVISION INFO:                                         **           
      *    *  04/07/14    CMB   CREATED                            **   CMB14097
      *    **********************************************************           
           MOVE 1                   TO WS-SPEC-IDX-IN                           
                                       WS-SPEC-IDX-OUT                          
           MOVE 0                   TO WS-SPEC-COUNT.                           
           MOVE WS-SPEC-STR-IN      TO WS-SPEC-STR-OUT.                         
      * EXPAND SS & ACCENTED CHARS IN THE GERMAN REGION                         
           IF WS-SPEC-CNTRY = 'DE' OR 'DC' OR 'AT' OR 'CH'                      
              PERFORM UNTIL WS-SPEC-IDX-OUT > WS-SPEC-MAX                       
                 EVALUATE WS-SPEC-CHAR-IN(WS-SPEC-IDX-IN)                       
                   WHEN '~'                                                     
                    MOVE 'S' TO WS-SPEC-CHAR-OUT(WS-SPEC-IDX-OUT)               
                    ADD 1 TO WS-SPEC-IDX-OUT                                    
                    MOVE 'S' TO WS-SPEC-CHAR-OUT(WS-SPEC-IDX-OUT)               
                    ADD +1 TO WS-SPEC-COUNT                                     
                   WHEN '{'                                                     
                    MOVE 'A' TO WS-SPEC-CHAR-OUT(WS-SPEC-IDX-OUT)               
                    ADD 1 TO WS-SPEC-IDX-OUT                                    
                    MOVE 'E' TO WS-SPEC-CHAR-OUT(WS-SPEC-IDX-OUT)               
                    ADD +1 TO WS-SPEC-COUNT                                     
                   WHEN '¢'                                                     
                    MOVE 'A' TO WS-SPEC-CHAR-OUT(WS-SPEC-IDX-OUT)               
                    ADD 1 TO WS-SPEC-IDX-OUT                                    
                    MOVE 'E' TO WS-SPEC-CHAR-OUT(WS-SPEC-IDX-OUT)               
                    ADD +1 TO WS-SPEC-COUNT                                     
                   WHEN '}'                                                     
                    MOVE 'U' TO WS-SPEC-CHAR-OUT(WS-SPEC-IDX-OUT)               
                    ADD 1 TO WS-SPEC-IDX-OUT                                    
                    MOVE 'E' TO WS-SPEC-CHAR-OUT(WS-SPEC-IDX-OUT)               
                    ADD +1 TO WS-SPEC-COUNT                                     
                   WHEN '!'                                                     
                    MOVE 'U' TO WS-SPEC-CHAR-OUT(WS-SPEC-IDX-OUT)               
                    ADD 1 TO WS-SPEC-IDX-OUT                                    
                    MOVE 'E' TO WS-SPEC-CHAR-OUT(WS-SPEC-IDX-OUT)               
                    ADD +1 TO WS-SPEC-COUNT                                     
                   WHEN '¦'                                                     
                    MOVE 'O' TO WS-SPEC-CHAR-OUT(WS-SPEC-IDX-OUT)               
                    ADD 1 TO WS-SPEC-IDX-OUT                                    
                    MOVE 'E' TO WS-SPEC-CHAR-OUT(WS-SPEC-IDX-OUT)               
                    ADD +1 TO WS-SPEC-COUNT                                     
                   WHEN '\'                                                     
                    MOVE 'O' TO WS-SPEC-CHAR-OUT(WS-SPEC-IDX-OUT)               
                    ADD 1 TO WS-SPEC-IDX-OUT                                    
                    MOVE 'E' TO WS-SPEC-CHAR-OUT(WS-SPEC-IDX-OUT)               
                    ADD +1 TO WS-SPEC-COUNT                                     
                   WHEN OTHER                                                   
                    MOVE WS-SPEC-CHAR-IN(WS-SPEC-IDX-IN) TO                     
                         WS-SPEC-CHAR-OUT(WS-SPEC-IDX-OUT)                      
                 END-EVALUATE                                                   
                 ADD 1 TO WS-SPEC-IDX-IN                                        
                          WS-SPEC-IDX-OUT                                       
              END-PERFORM.
