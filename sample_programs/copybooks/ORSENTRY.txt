       G9999-CHECK-ENTRY-FEAT-CODE.                                             
                                                                                
           MOVE ORH-COMPANY            TO     TB-ENTRY-COMPANY-CODE.            
           CALL 'SYU510'               USING  TB-ENTRY-PROC-IA.                 
                                                                                
           MOVE SPACES                 TO  WS-ENTRY-FEAT-SW.                    
                                                                                
           EVALUATE TRUE                                                        
               WHEN TB-ENTRY-WEB AND ORH-FROM-WEB                               
                   MOVE 'Y'            TO  WS-ENTRY-FEAT-SW                     
               WHEN TB-ENTRY-INSIDELINE AND ORH-FROM-IL                         
                   MOVE 'Y'            TO  WS-ENTRY-FEAT-SW                     
               WHEN TB-ENTRY-TELESALES AND ORH-ENTRY-ORDER-ENTRY                
                   MOVE 'Y'            TO  WS-ENTRY-FEAT-SW                     
               WHEN TB-ENTRY-EDI AND ORH-OFF-LINE-ENTRY                         
                   MOVE 'Y'            TO  WS-ENTRY-FEAT-SW                     
               WHEN TB-ENTRY-XML AND ORH-FROM-XML                               
                   MOVE 'Y'            TO  WS-ENTRY-FEAT-SW                     
               WHEN OTHER                                                       
                   MOVE 'N'            TO  WS-ENTRY-FEAT-SW                     
           END-EVALUATE.                                                        
                                                                                
       G9999-EXIT.                                                              
            EXIT.
